			PressedRadio = Interactor->GetFocusedActor();
			InteractPressedAt = GetWorld()->GetTimeSeconds();

void ARoadToWisdomOnFootCharacter::ReleaseInteract()
{
	if (!Interactor) return;
	Interactor->CancelHoldInteract();
	if (PressedRadio.IsValid() && PressedRadio.Get() == Interactor->GetFocusedActor()
		&& GetWorld()->GetTimeSeconds() - InteractPressedAt <= RadioTapSeconds) {
		Interactor->TryInteractOption(TEXT("Radio"));
	}
	PressedRadio.Reset();
}

void ARoadToWisdomOnFootCharacter::CancelInteract()

}

{
	if (SaveStableId.IsNone()) {
		UE_LOG(LogRoadToWisdom, Error, TEXT("Pickup %s needs a unique SaveStableId."), *GetName());
		return;
	}
	// TryPickup owns validation and transfer. Destroy marks the actor pending kill;
	// capture the coordinator/id before calling it, then use its accepted quantity.

		? GetGameInstance()->GetSubsystem<URoadToWisdomSaveCoordinator>()
		: nullptr;
	const FName Id = SaveStableId;
	const int32 Before = GetItem().GetCount();
	const int32 Accepted = TryPickup(Info.Interactor.Get());
	if (Coordinator && Accepted > 0 && Accepted == Before) {
		Coordinator->NotifyPickupCollected(Id);
	}
}

bool ARoadToWisdomFuseBox::CanInteract_Implementation(const FUTInteractionInfo& Info) const
{
	if (bFuseInstalled)
	{
		return false; // nothing offered anymore; prompts disappear via empty options
	}

	// Installing requires the fuse in the interacting pawn's inventory.
	const AActor* Interactor = Info.Interactor.Get();
	const UUTInventoryComponent* Inventory = Interactor ? Interactor->FindComponentByClass<UUTInventoryComponent>() : nullptr;
	return Inventory && Inventory->HasItems(FuseItemId, 1);
}

void ARoadToWisdomFuseBox::Interact_Implementation(const FUTInteractionInfo& Info)
{
	if (bFuseInstalled || !HasAuthority())
	{
		return;
	}

	AActor* Interactor = Info.Interactor.Get();
	UUTInventoryComponent* Inventory = Interactor ? Interactor->FindComponentByClass<UUTInventoryComponent>() : nullptr;
	if (!Inventory || !Inventory->ConsumeItems(FuseItemId, 1))
	{
		return;
	}

	bFuseInstalled = true;
	OnRep_FuseInstalled();
	OnFuseInstalled.Broadcast();

	UE_LOG(LogRoadToWisdom, Log, TEXT("FuseBox %s: fuse installed."), *GetSaveStableId_Implementation().ToString());
}

	GetWorldTimerManager().ClearTimer(ArrivalTimer);
	Super::EndPlay(EndPlayReason);
}

FText ARoadToWisdomMissionManager::GetRadioMessage() const
{
	switch (Stage) {
		case EQRRelayMissionStage::WaitForArrival: return NSLOCTEXT(
				"QuietRoad", "RadioStage0", "Тишина в эфире. Координаты ретранслятора помечены: восток, у дороги."
			);
		case EQRRelayMissionStage::FindFuse: return NSLOCTEXT(
				"QuietRoad", "RadioStage1", "Ретранслятор молчит из-за питания. Найдите предохранитель и вставьте его в щиток."
			);
		case EQRRelayMissionStage::PullLever: return NSLOCTEXT(
				"QuietRoad", "RadioStage2", "Питание почти восстановлено. Включите рубильник станции."
			);
		case EQRRelayMissionStage::Completed:
		default: return NSLOCTEXT("QuietRoad", "RadioStage3", "Ретранслятор в эфире. Радиосканер обновил карту района.");
	}
}

void ARoadToWisdomMissionManager::AdvanceStage()
{
	if (Stage >= EQRRelayMissionStage::Completed) {
		return;
	}

	Stage = static_cast<EQRRelayMissionStage>(static_cast<uint8>(Stage) + 1);
	SetObjectiveText();

	if (Stage == EQRRelayMissionStage::Completed) {
		OnMissionMessage.Broadcast(CompletionMessage);
		OnMissionCompleted.Broadcast();
	}

	// Arrival can occur while driving. Persist at the safe exit event instead of
	// reporting an expected moving-vehicle refusal as a disk failure.
	if (auto* Explorer = Cast<ARoadToWisdomExplorerController>(UGameplayStatics::GetPlayerController(this, 0)))
	{
		if (Explorer->GetVanPawn() && Explorer->GetVanPawn()->GetVelocity().Size() > 50.f)
		{
			return;
		}
	}
	// Persist after every committed advance so reloads cannot replay a step.
	if (URoadToWisdomSaveCoordinator* Coordinator = GetGameInstance()->GetSubsystem<URoadToWisdomSaveCoordinator>()) {
		if (!Coordinator->SaveSession(this)) {
			UE_LOG(LogRoadToWisdom, Error, TEXT("AdvanceStage: session save failed, progress may not persist."));
			if (APlayerController* PC = UGameplayStatics::GetPlayerController(this, 0)) {
				if (ARoadToWisdomExplorerController* Explorer = Cast<ARoadToWisdomExplorerController>(PC)) {
					Explorer->ShowHUDHint(NSLOCTEXT("QuietRoad", "SaveFailed", "Не удалось сохранить прогресс"));
				}
			}
		}
	}
}

void ARoadToWisdomMissionManager::SetObjectiveText()
{
	FText Objective;
	switch (Stage) {
		case EQRRelayMissionStage::WaitForArrival: Objective = ObjectiveWaitForArrival;
			break;
		case EQRRelayMissionStage::FindFuse: Objective = ObjectiveFindFuse;
			break;
		case EQRRelayMissionStage::PullLever: Objective = ObjectivePullLever;
			break;
		case EQRRelayMissionStage::Completed: Objective = ObjectiveCompleted;
			break;
	}
	if (Lever) Lever->bUsable = Stage == EQRRelayMissionStage::PullLever;
	CurrentObjective = Objective;

	OnObjectiveChanged.Broadcast(Objective);
	if (APlayerController* PC = UGameplayStatics::GetPlayerController(this, 0)) {
		if (ARoadToWisdomExplorerController* Explorer = Cast<ARoadToWisdomExplorerController>(PC)) {
			Explorer->SetHUDObjective(Objective);
		}
	}
}

void ARoadToWisdomMissionManager::HandleFuseInstalled()
{
	if (Stage == EQRRelayMissionStage::FindFuse) {
		AdvanceStage(); // FindFuse -> PullLever
	}
}

void ARoadToWisdomMissionManager::HandleLeverSwitched(AUTLever* InLever, bool bIsOn)
{
	// Stage guard keeps replays of the lever event (including a save restore) idempotent.
	if (bIsOn && Stage == EQRRelayMissionStage::PullLever) {
		AdvanceStage(); // PullLever -> Completed
	}
}

void ARoadToWisdomMissionManager::CheckArrival()
{
	if (Stage != EQRRelayMissionStage::WaitForArrival || !FuseBox) {
		GetWorldTimerManager().ClearTimer(ArrivalTimer);
		return;
	}

	const APawn* PlayerPawn = UGameplayStatics::GetPlayerPawn(this, 0);
	if (PlayerPawn && FVector::Dist(PlayerPawn->GetActorLocation(), FuseBox->GetActorLocation()) <= ArriveRadius) {
		GetWorldTimerManager().ClearTimer(ArrivalTimer);
		AdvanceStage(); // WaitForArrival -> FindFuse
		OnMissionMessage.Broadcast(NSLOCTEXT("QuietRoad", "ArrivedMsg", "Ретранслятор рядом. Осмотрите щиток у опоры."));
	}
}

void ARoadToWisdomMissionManager::ApplyState(EQRRelayMissionStage InStage, bool bLeverWasOn)
{
	Stage = InStage;

	// The WorldMechanics lever has no IUTSaveable contract yet (see the toolkit
	// integration matrix), so its state rides in the mission section. SetOn broadcasts
	// OnSwitched, but HandleLeverSwitched is stage-guarded and no-ops after a restore.
	if (Lever) {
		Lever->bIsOn = bLeverWasOn;
		if (Lever->HandleMesh) Lever->HandleMesh->SetRelativeRotation(bLeverWasOn ? Lever->HandleOnRotation : FRotator::ZeroRotator);
	}

	SetObjectiveText();

	if (Stage != EQRRelayMissionStage::WaitForArrival) {
		GetWorldTimerManager().ClearTimer(ArrivalTimer);
	}
}

void ARoadToWisdomExplorerController::BeginPlay()
{

	ARoadToWisdomPlayerController::BeginPlay();

	if (IsLocalPlayerController() && !HUD)
	{
		HUD = CreateWidget<URoadToWisdomHUDWidget>(this, HUDWidgetClass);
		if (HUD)
		{
			HUD->AddToPlayerScreen(0);
		}
		else
		{
			UE_LOG(LogRoadToWisdom, Error, TEXT("ExplorerController: could not create the HUD widget."));
		}
	}

	// The game starts on foot, so the placed van never passes through OnPossess here:

void ARoadToWisdomExplorerController::RegisterVanFromWorld()
{
	if (bVanRegistrationAttempted || IsValid(VanPawn))
	{
		return;
	}
	bVanRegistrationAttempted = true;

	TArray<AActor*> Tagged;
	UGameplayStatics::GetAllActorsWithTag(this, VanTag, Tagged);

	if (Tagged.Num() == 0)
	{
		// Template maps have no tagged van; the slice map must have exactly one.
		UE_LOG(LogRoadToWisdom, Warning, TEXT("RegisterVanFromWorld: no actor tagged '%s' in this map."), *VanTag.ToString());
		return;
	}
	if (Tagged.Num() > 1)
	{
		// Ambiguity blocks registration entirely (GDD v0.3 28.1): picking an arbitrary
		// van would leave the radio/controller disagreeing about which actor is real.
		UE_LOG(LogRoadToWisdom, Error, TEXT("RegisterVanFromWorld: %d actors tagged '%s', exactly one expected. Van stays unregistered; F enter/exit disabled until fixed."), Tagged.Num(), *VanTag.ToString());
		return;
	}

	VanPawn = Cast<ARoadToWisdomPawn>(Tagged[0]);
	if (!IsValid(VanPawn))
	{
		UE_LOG(LogRoadToWisdom, Error, TEXT("RegisterVanFromWorld: tagged actor '%s' is not an ARoadToWisdomPawn."), *Tagged[0]->GetName());
	}
}

	TArray<AActor*> Managers;
	UGameplayStatics::GetAllActorsOfClass(this, ARoadToWisdomMissionManager::StaticClass(), Managers);
	if (Managers.Num() == 0)
	{

	if (Managers.Num() > 1)

	}

	if (const ARoadToWisdomMissionManager* Manager = Cast<ARoadToWisdomMissionManager>(Managers[0]))

	}
}

void ARoadToWisdomExplorerController::OnPossess(APawn* InPawn)
{
	// Template OnPossess CastChecks the pawn into ARoadToWisdomPawn and would assert
	// when possessing the on-foot character, so the base implementation is used and
	// the vehicle bookkeeping is redone here.
	APlayerController::OnPossess(InPawn);

	if (ARoadToWisdomPawn* Vehicle = Cast<ARoadToWisdomPawn>(InPawn))
	{
		// The template's vehicle respawn path (VehiclePawnClass) is not configured here:
		// the van is a persistent placed actor for the slice, not a respawnable default pawn.
		VanPawn = Vehicle;
		ControlMode = EQRControlMode::Driving;
		ApplyMappingContexts(EQRControlMode::Driving);
		UpdateHudMode();
	}
	else if (ARoadToWisdomOnFootCharacter* Explorer = Cast<ARoadToWisdomOnFootCharacter>(InPawn))
	{
		OnFootCharacter = Explorer;
		Explorer->SetVehicleStowed(false);
		ControlMode = EQRControlMode::OnFoot;
		ApplyMappingContexts(EQRControlMode::OnFoot);
		UpdateHudMode();

		if (bHasPendingRestore)
		{
			// Apply outside the possession stack: the restore may switch possession
			// to the saved van, and nested Possess calls inside OnPossess are fragile.
			GetWorldTimerManager().SetTimerForNextTick(FTimerDelegate::CreateWeakLambda(this, [this]()
			{
				ApplyPendingSessionRestore();
			}));
		}
	}
}

	APlayerController::OnUnPossess();
}

void ARoadToWisdomExplorerController::RequestVehicleToggle()
{
	if (bVehicleTransitionRunning)
	{
		return;
	}

	bVehicleTransitionRunning = true;
	bool bChanged = false;
	if (ControlMode == EQRControlMode::Driving)
	{
		bChanged = TryExitVehicle();
	}
	else
	{
		bChanged = TryEnterVehicle();

	{
		// Failure reasons are already shown by the Try* functions as HUD hints.
	}
}

bool ARoadToWisdomExplorerController::TryExitVehicle()
{
	if (!VanPawn)
	{
		UE_LOG(LogRoadToWisdom, Warning, TEXT("TryExitVehicle: no van pawn (template-only map?)."));
		return false;
	}

	if (VanPawn->GetVelocity().Size() > ExitMaxSpeed)
	{
		ShowHUDHint(NSLOCTEXT("QuietRoad", "StopFirst", "Остановите фургон, чтобы выйти"));
		return false;
	}

	const FRotator VehicleRotation = VanPawn->GetActorRotation();
	FVector ExitLocation = FVector::ZeroVector;
	bool bFoundSpot = false;

	for (const FVector& Offset : ExitOffsets)
	{
		const FVector Candidate = VanPawn->GetActorTransform().TransformPosition(Offset);

		// Drop the candidate to the ground so the character never spawns in the air.
		FHitResult GroundHit;
		const FVector TraceStart(Candidate.X, Candidate.Y, Candidate.Z + 300.f);
		const FVector TraceEnd(Candidate.X, Candidate.Y, Candidate.Z - 300.f);
		if (!UKismetSystemLibrary::LineTraceSingle(this, TraceStart, TraceEnd, UEngineTypes::ConvertToTraceType(ECC_Visibility), false,
			TArray<AActor*>{VanPawn}, EDrawDebugTrace::None, GroundHit, true))
		{
			continue;
		}

		const FVector Grounded = GroundHit.Location + FVector(0.f, 0.f, 90.f + 5.f);
		if (!IsExitSpotClear(FTransform(VehicleRotation, Grounded)))
		{
			continue;
		}

		ExitLocation = Grounded;
		bFoundSpot = true;
		break;
	}

	if (!bFoundSpot)
	{
		ShowHUDHint(NSLOCTEXT("QuietRoad", "NoExitRoom", "Вокруг фургона нет места для выхода"));
		return false;
	}

	if (!OnFootCharacter)
	{
		FActorSpawnParameters Params;
		Params.Owner = this;
		Params.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AdjustIfPossibleButAlwaysSpawn;
		OnFootCharacter = GetWorld()->SpawnActor<ARoadToWisdomOnFootCharacter>(OnFootCharacterClass, FTransform(VehicleRotation, ExitLocation), Params);
		if (!OnFootCharacter)
		{
			ShowHUDHint(NSLOCTEXT("QuietRoad", "SpawnFailed", "Не удалось создать персонажа"));
			return false;
		}
	}
	else
	{
		OnFootCharacter->SetActorLocationAndRotation(ExitLocation, VehicleRotation, false, nullptr, ETeleportType::ResetPhysics);
	}

	// Park the van: no input leaks into the physics while the player walks away.
	VanPawn->DoThrottle(0.f);
	VanPawn->DoBrakeStop();
	VanPawn->DoHandbrakeStart();

	Possess(OnFootCharacter); // OnPossess applies mode + mapping contexts

	// Save the safe parked state after deferred arrival progress, using the same
	// coordinator as the radio (no timer/polling save path).
	for (TActorIterator<ARoadToWisdomMissionManager> It(GetWorld()); It; ++It)
	{
		if (auto* Save = GetGameInstance()->GetSubsystem<URoadToWisdomSaveCoordinator>())
			Save->SaveSession(*It);
		break;
	}

	ShowHUDHint(NSLOCTEXT("QuietRoad", "OnFootHint", "E — действие у объекта, F — сесть обратно"));
	return true;
}

bool ARoadToWisdomExplorerController::TryEnterVehicle()
{
	if (!VanPawn)
	{
		return false;
	}

	// Measure from the driver door, not from the vehicle origin: a long camper van
	// would otherwise accept an F from the middle of its body.
	const FVector DoorLocation = VanPawn->GetActorTransform().TransformPosition(EnterProbeOffset);
	if (!OnFootCharacter || FVector::Dist(OnFootCharacter->GetActorLocation(), DoorLocation) > EnterMaxDistance)
	{
		ShowHUDHint(NSLOCTEXT("QuietRoad", "ComeCloser", "Подойдите к водительской двери, чтобы сесть"));
		return false;
	}

	Possess(VanPawn); // OnPossess applies mode + mapping contexts
	VanPawn->DoHandbrakeStop();

	if (OnFootCharacter)
	{
		OnFootCharacter->SetVehicleStowed(true);
	}
	return true;
}

void ARoadToWisdomExplorerController::ApplyMappingContexts(EQRControlMode Mode)
{
	UEnhancedInputLocalPlayerSubsystem* Subsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(GetLocalPlayer());
	if (!Subsystem)
	{
		return;
	}

	const TArray<TObjectPtr<UInputMappingContext>>& ToAdd = (Mode == EQRControlMode::Driving) ? VehicleMappingContexts : OnFootMappingContexts;
	const TArray<TObjectPtr<UInputMappingContext>>& ToRemove = (Mode == EQRControlMode::Driving) ? OnFootMappingContexts : VehicleMappingContexts;

	for (const TObjectPtr<UInputMappingContext>& Context : ToRemove)
	{
		if (Context)
		{
			Subsystem->RemoveMappingContext(Context);
		}
	}
	for (const TObjectPtr<UInputMappingContext>& Context : ToAdd)
	{
		if (Context)
		{
			Subsystem->AddMappingContext(Context, 0);
		}
	}
}

bool ARoadToWisdomExplorerController::IsExitSpotClear(const FTransform& WorldTransform) const
{
	FHitResult Hit;
	TArray<AActor*> Ignored;
	if (VanPawn)
	{
		Ignored.Add(VanPawn);
	}
	if (OnFootCharacter)
	{
		Ignored.Add(OnFootCharacter);
	}

	// Full capsule volume of the on-foot pawn. WorldTransform is the pawn's origin
	// (capsule center); lift the trace 2 cm so the ground the candidate stands on
	// does not count as an obstruction while walls/trees still do.
	const FVector Center = WorldTransform.GetLocation() + FVector(0.f, 0.f, 2.f);
	return !UKismetSystemLibrary::CapsuleTraceSingle(this, Center, Center,
		ExitSpotRadius, 90.f, UEngineTypes::ConvertToTraceType(ECC_Pawn), false, Ignored, EDrawDebugTrace::None, Hit, true);
}

void ARoadToWisdomExplorerController::ApplyPendingSessionRestore()
{
	if (!bHasPendingRestore)
	{
		return;
	}
	bHasPendingRestore = false;

	if (!OnFootCharacter)
	{
		return;
	}

	if (bHasPendingInventory && OnFootCharacter->GetInventory())
	{
		OnFootCharacter->GetInventory()->RestoreFromSnapshot(PendingInventory); // silent, no events
		bHasPendingInventory = false;
	}

	if (!PendingPlayerSection.bValid)
	{
		return;
	}

	if (PendingPlayerSection.bOnFoot)
	{
		OnFootCharacter->SetActorLocationAndRotation(PendingPlayerSection.Transform.GetLocation(), PendingPlayerSection.Transform.Rotator(),
			false, nullptr, ETeleportType::ResetPhysics);
	}
	else if (VanPawn)
	{
		// Saved while driving: hand the van back instead of spawning at the start spot.
		Possess(VanPawn);
		VanPawn->DoHandbrakeStop();
		if (OnFootCharacter)
		{
			OnFootCharacter->SetVehicleStowed(true);
		}
	}
}

void ARoadToWisdomExplorerController::TryRestoreSession()
{
	if (bSessionRestoreAttempted)
	{
		return;
	}
	bSessionRestoreAttempted = true;

	if (URoadToWisdomSaveCoordinator* Coordinator = GetGameInstance()->GetSubsystem<URoadToWisdomSaveCoordinator>())
	{
		Coordinator->TryRestoreSession(this);
	}

	// Possession may have happened before this BeginPlay ran (auto-possessed local
	// players); OnPossess's next-tick hook will never fire for it. Apply now — this
	// path is outside any possession stack. The guard inside keeps it one-shot.
	if (bHasPendingRestore && OnFootCharacter)
	{
		ApplyPendingSessionRestore();
	}
}

bool URoadToWisdomSaveCoordinator::SaveSession(ARoadToWisdomMissionManager* Mission)
{
	UWorld* World = GetGameInstance()->GetWorld();
	auto* Controller = Cast<ARoadToWisdomExplorerController>(UGameplayStatics::GetPlayerController(World, 0));
	auto* Save = GetGameInstance()->GetSubsystem<UUTSaveSubsystem>();
	if (!World || !Mission || !Controller || !Save || !CaptureAndWriteSections(Mission, Controller)) {
		UE_LOG(LogRoadToWisdom, Warning, TEXT("SaveSession: required state unavailable/invalid; slot not written."));
		return false;
	}
	// UTSave delegates to SaveGameToSlot. Its transactional LOAD protects Current;
	// neither section writes nor actor replay promise power-loss-safe disk writes.
	const FUTResult Result = Save->SaveToSlot(GetSessionSlotName());
	UE_LOG(LogRoadToWisdom, Log, TEXT("SaveSession: success=%d %s"), Result.IsSuccess(), *Result.Message.ToString());
	return Result.IsSuccess();
}

bool URoadToWisdomSaveCoordinator::CaptureAndWriteSections(
	ARoadToWisdomMissionManager* Mission, ARoadToWisdomExplorerController* Controller
)
{
	UWorld* World = GetGameInstance()->GetWorld();
	auto* Save = GetGameInstance()->GetSubsystem<UUTSaveSubsystem>();
	auto* WorldState = World->GetSubsystem<UUTWorldStateSubsystem>();
	if (!WorldState || !Mission->FuseBox || !Mission->Lever || !Controller->VanPawn
		|| !Controller->OnFootCharacter || !Controller->OnFootCharacter->GetInventory()
		|| !Controller->GetPawn() || Controller->bHasPendingRestore
		|| Controller->VanPawn->GetVelocity().Size() > Controller->ExitMaxSpeed)
		return false;
	TMap<FName, ARoadToWisdomPlacedPickup*> ById;
	if (!GatherPickups(World, ById)) return false;
	FQRPickupsSection Pickups;
	Pickups.CollectedIds = CollectedPickupIds;
	for (const auto& Pair : ById) {
		if (CollectedPickupIds.Contains(Pair.Key) || !Pair.Value->GetItem().IsValid()) return false;
		auto& Entry = Pickups.Remaining.AddDefaulted_GetRef();
		Entry.StableId = Pair.Key;
		Entry.Item = Pair.Value->GetItem();
	}
	FUTWorldStateSnapshot Snapshot;
	if (!WorldState->CaptureWorldState(Snapshot).IsSuccess() || Snapshot.Entries.IsEmpty()) return false;
	FQRMissionSection MissionSection;
	MissionSection.Stage = static_cast<int32>(Mission->GetStage());
	MissionSection.bLeverOn = Mission->Lever->bIsOn;
	FQRVehicleSection Vehicle;
	Vehicle.Transform = Controller->VanPawn->GetActorTransform();
	Vehicle.bValid = ValidPlacement(Vehicle.Transform);
	FQRPlayerSection Player;
	Player.Transform = Controller->GetPawn()->GetActorTransform();
	Player.bOnFoot = !Controller->IsDriving();
	Player.bValid = ValidPlacement(Player.Transform);
	if (!Vehicle.bValid || !Player.bValid) return false;
	// Capture all mandatory components first. Every save overwrites all six sections.
	return Save->WriteSection(TEXT("Mission"), Version, MissionSection).IsSuccess()
		&& Save->WriteSection(TEXT("Inventory"), Version, Controller->OnFootCharacter->GetInventory()->MakeSnapshot()).IsSuccess()
		&& Save->WriteSection(TEXT("World"), Version, Snapshot).IsSuccess()
		&& Save->WriteSection(TEXT("Pickups"), PickupsVersion, Pickups).IsSuccess()
		&& Save->WriteSection(TEXT("Vehicle"), Version, Vehicle).IsSuccess()
		&& Save->WriteSection(TEXT("Player"), Version, Player).IsSuccess();
}

bool URoadToWisdomSaveCoordinator::TryRestoreSession(ARoadToWisdomExplorerController* Controller)
{
	// GameInstance survives map travel: never retain the previous session's history.
	CollectedPickupIds.Reset();
	if (Controller) {
		Controller->bHasPendingRestore = false;
		Controller->bHasPendingInventory = false;
		Controller->PendingInventory = FUTInventorySnapshot();
	}
	auto* Save = GetGameInstance()->GetSubsystem<UUTSaveSubsystem>();
	UWorld* World = GetGameInstance()->GetWorld();
	if (!Save || !World || !Controller || !Controller->VanPawn) return false;
	if (!Save->LoadFromSlot(GetSessionSlotName()).IsSuccess()) return false;
	FQRMissionSection MissionSection;
	FQRVehicleSection Vehicle;
	FQRPlayerSection Player;
	FQRPickupsSection Pickups;
	FUTInventorySnapshot Inventory;
	FUTWorldStateSnapshot Snapshot;
	// Read all data before applying ANY gameplay state.
	if (!Save->ReadSection(TEXT("Mission"), Version, MissionSection).IsSuccess()
		|| !Save->ReadSection(TEXT("Vehicle"), Version, Vehicle).IsSuccess()
		|| !Save->ReadSection(TEXT("Player"), Version, Player).IsSuccess()
		|| !Save->ReadSection(TEXT("Inventory"), Version, Inventory).IsSuccess()
		|| !Save->ReadSection(TEXT("World"), Version, Snapshot).IsSuccess()
		|| !Save->ReadSection(TEXT("Pickups"), PickupsVersion, Pickups).IsSuccess()) {
		UE_LOG(
			LogRoadToWisdom, Warning,
			TEXT("Restore rejected: missing/corrupt/obsolete section (Pickups requires v2). World unchanged.")
		);
		return false;
	}
	auto* WorldState = World->GetSubsystem<UUTWorldStateSubsystem>();
	TArray<AActor*> Managers;
	UGameplayStatics::GetAllActorsOfClass(World, ARoadToWisdomMissionManager::StaticClass(), Managers);
	if (!WorldState || Managers.Num() != 1 || !Vehicle.bValid || !Player.bValid
		|| !ValidPlacement(Vehicle.Transform) || !ValidPlacement(Player.Transform)
		|| MissionSection.Stage < 0 || MissionSection.Stage > static_cast<int32>(EQRRelayMissionStage::Completed))
		return false;
	auto* Mission = CastChecked<ARoadToWisdomMissionManager>(Managers[0]);
	if (!Mission->FuseBox || !Mission->Lever) return false;
	// This slice owns exactly one saveable type. UT ApplyWorldState is sequential,
	// so validate its identity, schema and stage consistency before invoking it.
	FUTWorldStateSnapshot Current;
	if (!WorldState->CaptureWorldState(Current).IsSuccess() || Current.Entries.Num() != 1 || Snapshot.Entries.Num() != 1)
		return
			false;
	const auto& Entry = Snapshot.Entries[0];
	const auto* Fuse = Entry.State.GetPtr<FQRFuseBoxState>();
	if (!Fuse || Entry.StableId != Mission->FuseBox->GetSaveStableId_Implementation()
		|| Entry.StableId != Current.Entries[0].StableId)
		return false;
	if (Fuse->bFuseInstalled != (MissionSection.Stage >= static_cast<int32>(EQRRelayMissionStage::PullLever))
		|| MissionSection.bLeverOn != (MissionSection.Stage == static_cast<int32>(EQRRelayMissionStage::Completed)))
		return false;
	const auto* Defaults = GetDefault<ARoadToWisdomOnFootCharacter>()->GetInventory();
	if (!Defaults || (Defaults->MaxSlots > 0 && Inventory.Stacks.Num() + Inventory.Instances.Num() > Defaults->MaxSlots))
		return
			false;
	TSet<FGuid> InstanceIds;
	int64 FuseCount = 0;
	for (const auto& Stack : Inventory.Stacks) {
		if (Stack.ItemId.IsNone() || Stack.Count <= 0) return false;
		if (Stack.ItemId == Mission->FuseBox->FuseItemId) FuseCount += Stack.Count;
	}
	for (const auto& Instance : Inventory.Instances) {
		if (!Instance.IsValid() || InstanceIds.Contains(Instance.InstanceId)) return false;
		InstanceIds.Add(Instance.InstanceId);
	}
	TMap<FName, ARoadToWisdomPlacedPickup*> ById;
	if (!GatherPickups(World, ById)) return false;
	TSet<FName> Seen;
	for (FName Id : Pickups.CollectedIds) {
		if (Id.IsNone() || Seen.Contains(Id) || !ById.Contains(Id)) return false;
		Seen.Add(Id);
	}
	for (const auto& Remaining : Pickups.Remaining) {
		auto** Target = ById.Find(Remaining.StableId);
		if (!Target || Seen.Contains(Remaining.StableId) || !Remaining.Item.IsValid()
			|| Remaining.Item.GetItemId() != (*Target)->InitialItem.GetItemId()
			|| Remaining.Item.bUnique != (*Target)->InitialItem.bUnique
			|| Remaining.Item.GetCount() > (*Target)->InitialItem.GetCount())
			return false;
		Seen.Add(Remaining.StableId);
		if (Remaining.Item.GetItemId() == Mission->FuseBox->FuseItemId) FuseCount += Remaining.Item.GetCount();
	}
	if (Seen.Num() != ById.Num() || FuseCount + (Fuse->bFuseInstalled ? 1 : 0) != 1) return false;
	// AUTWorldItem refuses payload reassignment. For partial stacks, prepare a fresh
	// authored actor through the supported initialization path, retaining originals
	// until every spawn succeeds. No item-transfer logic is copied from Toolkit.
	TArray<ARoadToWisdomPlacedPickup*> Prepared, Replaced;
	for (const auto& Remaining : Pickups.Remaining) {
		auto* Original = ById[Remaining.StableId];
		if (!Remaining.Item.bUnique && Remaining.Item.GetCount() == Original->GetItem().GetCount()) continue;
		FActorSpawnParameters Params;
		Params.bDeferConstruction = true;
		Params.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AlwaysSpawn;
		auto* Replacement = World->SpawnActor<ARoadToWisdomPlacedPickup>(
			Original->GetClass(), Original->GetActorTransform(), Params
		);
		if (!Replacement || !Replacement->InitializeItem(Remaining.Item)) {
			if (Replacement) Replacement->Destroy();
			for (auto* Pending : Prepared) Pending->Destroy();
			return false;
		}
		Replacement->SaveStableId = Original->SaveStableId;
		Replacement->InitialItem = Original->InitialItem;
		Replacement->PickupDistance = Original->PickupDistance;
		Replacement->Mesh->SetStaticMesh(Original->Mesh->GetStaticMesh());
		Replacement->Mesh->SetRelativeScale3D(Original->Mesh->GetRelativeScale3D());
		for (int32 I = 0; I < Original->Mesh->GetNumMaterials(); ++I)
			Replacement->Mesh->SetMaterial(
				I, Original->Mesh->GetMaterial(I)
			);
		Prepared.Add(Replacement);
		Replaced.Add(Original);
	}
	FUTResult ApplyResult;
	const int32 Applied = WorldState->ApplyWorldState(Snapshot, ApplyResult);
	if (!ApplyResult.IsSuccess() || Applied != Snapshot.Entries.Num()) {
		for (auto* Pending : Prepared) Pending->Destroy();
		UE_LOG(LogRoadToWisdom, Error, TEXT("WorldState rejected (%d entries): %s"), Applied, *ApplyResult.Message.ToString());
		return false;
	}
	for (int32 I = 0; I < Prepared.Num(); ++I) {
		const FTransform Transform = Replaced[I]->GetActorTransform();
		Replaced[I]->Destroy();
		Prepared[I]->FinishSpawning(Transform);
	}
	CollectedPickupIds = Pickups.CollectedIds;
	DestroyCollectedPickups();
	Mission->ApplyState(static_cast<EQRRelayMissionStage>(MissionSection.Stage), MissionSection.bLeverOn);
	Controller->VanPawn->SetActorLocationAndRotation(
		Vehicle.Transform.GetLocation(), Vehicle.Transform.Rotator(), false, nullptr, ETeleportType::TeleportPhysics
	);
	Controller->VanPawn->GetMesh()->SetPhysicsLinearVelocity(FVector::ZeroVector);
	Controller->VanPawn->GetMesh()->SetPhysicsAngularVelocityInDegrees(FVector::ZeroVector);
	Controller->VanPawn->DoThrottle(0.f);
	Controller->VanPawn->DoHandbrakeStart();
	Controller->PendingPlayerSection = Player;
	Controller->PendingInventory = Inventory;
	Controller->bHasPendingInventory = true;
	Controller->bHasPendingRestore = true;
	UE_LOG(LogRoadToWisdom, Log, TEXT("RestoreSession: validated six sections, applied %d world entries."), Applied);
	return true;
}

/**

 *  world capture/restore on the authority, slot orchestration in game code).
 *
 *  Sections: "Mission" (stage + lever), "Inventory" (UT inventory snapshot),
 *  "World" (UT world-state snapshot of all IUTSaveable actors), "Vehicle" and
 *  "Player" (placement + mode). Saved at mission stage commits; restored once from
 *  BeginPlay of the explorer controller. Single-player slice: direct synchronous
 *  calls on the game thread.
 */
UCLASS()
class URoadToWisdomSaveCoordinator : public UGameInstanceSubsystem
{
	GENERATED_BODY()

public:
	static constexpr TCHAR SlotName[] = TEXT("QuietRoad01");

	static FName GetSessionSlotName();

	/** Runtime record of a picked-up placed pickup (called by ARoadToWisdomPlacedPickup). */
	void NotifyPickupCollected(FName StableId);

	/** Restore-time sweep: destroys placed pickups whose ids were collected before the save. */
	void DestroyCollectedPickups() const;

	/**
	 * Capture mission stage, inventory, world state and placements, then flush to

	SetRootComponent(PanelMesh);

void ARoadToWisdomVanRadioPanel::BeginPlay()
{
	Super::BeginPlay();

	// Attach to the one tagged van so the panel rides along. Same registration
	// contract as the controller: exactly one match is the accepted state.
	TArray<AActor*> Tagged;
	UGameplayStatics::GetAllActorsWithTag(this, VanTag, Tagged);
	if (Tagged.Num() == 1) {
		AttachToActor(Tagged[0], FAttachmentTransformRules::KeepWorldTransform);
	}
	else {
		UE_LOG(
			LogRoadToWisdom, Error, TEXT("VanRadioPanel: %d actors tagged '%s', expected exactly 1; panel stays placed."),
			Tagged.Num(), *VanTag.ToString()
		);
	}
}

bool ARoadToWisdomVanRadioPanel::CanInteract_Implementation(const FUTInteractionInfo& Info) const
{
	const AActor* User = Info.Interactor.Get();
	return User && User->GetWorld() == GetWorld()
		&& FVector::DistSquared(User->GetActorLocation(), GetActorLocation()) <= FMath::Square(UseDistance);
}

void ARoadToWisdomVanRadioPanel::Interact_Implementation(const FUTInteractionInfo& Info)
{
	UWorld* World = GetWorld();
	if (!World || !HasAuthority() || !CanInteract_Implementation(Info) || (Info.OptionId != TEXT("Radio") && Info.OptionId !=
		TEXT("Save"))) {
		return;
	}

	ARoadToWisdomExplorerController* Explorer = Cast<ARoadToWisdomExplorerController>(
		UGameplayStatics::GetPlayerController(World, 0)
	);
	if (!Explorer) {
		return;
	}

	TArray<AActor*> Managers;
	UGameplayStatics::GetAllActorsOfClass(World, ARoadToWisdomMissionManager::StaticClass(), Managers);
	ARoadToWisdomMissionManager* Mission = Managers.Num() > 0 ? Cast<ARoadToWisdomMissionManager>(Managers[0]) : nullptr;

	if (Info.OptionId == TEXT("Save")) {
		if (!Mission) {
			ShowHudMessage(NSLOCTEXT("QuietRoad", "SaveNoMission", "Сохранение недоступно: нет активной миссии"), 3.f);
			return;
		}
		if (URoadToWisdomSaveCoordinator* Coordinator = Explorer->GetGameInstance()->GetSubsystem<
			URoadToWisdomSaveCoordinator>()) {
			// Show success only on the real FUTResult outcome (GDD §23.1).
			if (Coordinator->SaveSession(Mission)) {
				ShowHudMessage(NSLOCTEXT("QuietRoad", "SaveDone", "Прогресс сохранён"), 3.f);
			}
			else {
				ShowHudMessage(NSLOCTEXT("QuietRoad", "SaveFailed", "Не удалось сохранить прогресс"), 4.f);
			}
		}
		return;
	}

	// Radio: objective plus a stage-dependent line (GDD §23.1 — plain text, no simulation).
	if (Mission) {
		ShowHudMessage(Mission->GetRadioMessage(), 8.f);
	}
	else {
		ShowHudMessage(NSLOCTEXT("QuietRoad", "RadioQuiet", "В эфире только шум ветра…"), 5.f);
	}
}

FText ARoadToWisdomVanRadioPanel::GetInteractionPrompt_Implementation(const FUTInteractionInfo& Info) const
{
	return NSLOCTEXT("QuietRoad", "RadioPrompt", "E — радио, удерживать — сохранить");
}

void ARoadToWisdomVanRadioPanel::GetInteractionOptions_Implementation(
	const FUTInteractionInfo& Info, TArray<FUTInteractionOption>& OutOptions
) const
{
	FUTInteractionOption Radio;
	Radio.OptionId = TEXT("Radio");
	Radio.Prompt = NSLOCTEXT("QuietRoad", "RadioOption", "E — радио");
	OutOptions.Add(Radio);

	FUTInteractionOption Save;
	Save.OptionId = TEXT("Save");
	Save.Prompt = NSLOCTEXT("QuietRoad", "SaveOption", "Удерживайте E — сохранить прогресс");
	Save.HoldDuration = SaveHoldSeconds;
	OutOptions.Add(Save);
}

int32 ARoadToWisdomVanRadioPanel::GetInteractionPriority_Implementation() const
{
	// Below mission targets (fuse box = 10) so it never steals focus in front of them.
	return 5;
}

		}
	}
}

void ARoadToWisdomDecorCluster::SetInstances(int32 Slot, const TArray<FTransform>& Instances)
{
	UHierarchicalInstancedStaticMeshComponent* Target = nullptr;
	switch (Slot) {
		case 0: Target = TreeTrunks;
			break;
		case 1: Target = TreeCrowns;
			break;
		case 2: Target = Bushes;
			break;
		case 3: Target = Rocks;
			break;
		default:
			UE_LOG(LogRoadToWisdom, Warning, TEXT("DecorCluster: unknown slot %d"), Slot);
			return;
	}

	// Official API only: both AddInstance and the batch AddInstances write straight
	// into the bulk-serialized PerInstanceSMData (UE 5.8 InstancedStaticMesh.cpp),
	// verified by a reload check after Save Level. The earlier direct-array write is
	// unnecessary; the actual persistence bug was scripts editing a stale current
	// world (now fixed by load_level in the generator).
	Target->Modify();
	Target->ClearInstances();
	for (const FTransform& Transform : Instances) {
		Target->AddInstance(Transform, true);
	}
	UE_LOG(LogRoadToWisdom, Log, TEXT("DecorCluster slot %d: %d instances written"), Slot, Target->GetNumInstances());
}

// Route acceptance driver: in a -game session, possesses the tagged van and
// follows the "Main" road polyline from Saved/WorldGeneration/World.json with a
// simple lookahead autopilot, then writes Saved/WorldGeneration/VanRouteMain.json
// and exits. Console command: QRW.RouteDrive

#include "CoreMinimal.h"
#include "Containers/Ticker.h"
#include "Engine/World.h"
#include "Engine/GameEngine.h"
#include "EngineUtils.h"
#include "GameFramework/PlayerController.h"
#include "ChaosWheeledVehicleMovementComponent.h"
#include "HAL/PlatformProcess.h"
#include "HAL/IConsoleManager.h"
#include "Misc/FileHelper.h"
#include "Misc/Paths.h"
#include "Dom/JsonObject.h"
#include "Serialization/JsonReader.h"
#include "Serialization/JsonSerializer.h"

#include "RoadToWisdom.h"

namespace QRWRouteDrive
{
	constexpr float WaypointRadiusCm = 2200.0f;   // advance to the next point inside 22 m
	constexpr float StallSpeedCm = 40.0f;         // below this under throttle...
	constexpr float StallTimeoutS = 8.0f;         // ...for this long = stuck
	constexpr float MaxRunSeconds = 900.0f;
	constexpr float OffroadLimitCm = 750.0f;      // 4.5 m half width + shoulder

	struct FRouteState
	{
		TArray<FVector> Points;
		float RouteLengthCm = 0.0f;
		int32 Waypoint = 1;
		double StartSeconds = 0.0;
		double StallSince = -1.0;
		int32 OffroadSamples = 0;
		int32 Samples = 0;
		double SpeedSum = 0.0;
		float SpeedMax = 0.0f;
		bool bFinished = false;
	};

	static FRouteState State;
	static FTSTicker::FDelegateHandle TickHandle;

	static float RouteLengthM()
	{
		return State.RouteLengthCm / 100.0f;
	}

	static void Finish(UWorld* World, const FString& Reason)
	{
		if (State.bFinished)
		{
			return;
		}
		State.bFinished = true;

		const double Elapsed = FPlatformTime::Seconds() - State.StartSeconds;
		const float Progress = static_cast<float>(State.Waypoint) / FMath::Max(1, State.Points.Num() - 1);
		float ArcCm = 0.0f;
		for (int32 i = 0; i < State.Waypoint && i + 1 < State.Points.Num(); ++i)
		{
			ArcCm += FVector::Dist2D(State.Points[i], State.Points[i + 1]);
		}
		const float AvgKmh = State.Samples > 0 ? (State.SpeedSum / State.Samples) * 0.036f : 0.0f;
		const float MaxKmh = State.SpeedMax * 0.036f;

		TSharedPtr<FJsonObject> Root = MakeShared<FJsonObject>();
		Root->SetStringField(TEXT("reason"), Reason);
		Root->SetNumberField(TEXT("route_points"), State.Points.Num());
		Root->SetNumberField(TEXT("route_length_m"), RouteLengthM());
		Root->SetNumberField(TEXT("reached_waypoint"), State.Waypoint);
		Root->SetNumberField(TEXT("progress"), Progress);
		Root->SetNumberField(TEXT("travelled_m"), ArcCm / 100.0);
		Root->SetNumberField(TEXT("elapsed_s"), Elapsed);
		Root->SetNumberField(TEXT("avg_kmh"), AvgKmh);
		Root->SetNumberField(TEXT("max_kmh"), MaxKmh);
		Root->SetNumberField(TEXT("offroad_samples_over_limit"), State.OffroadSamples);
		Root->SetNumberField(TEXT("samples"), State.Samples);
		Root->SetBoolField(TEXT("passed"),
			Reason == TEXT("completed") && Progress >= 0.95f && AvgKmh >= 12.0f &&
			State.OffroadSamples <= State.Samples * 0.05f);

		FString Output;
		const auto Writer = TJsonWriterFactory<>::Create(&Output);
		FJsonSerializer::Serialize(Root.ToSharedRef(), Writer);
		const FString Path = FPaths::ProjectSavedDir() / TEXT("WorldGeneration/VanRouteMain.json");
		FFileHelper::SaveStringToFile(Output, *Path);

		UE_LOG(LogRoadToWisdom, Warning, TEXT("[ROUTE-RESULT] %s"), *Output);

		if (TickHandle.IsValid())
		{
			FTSTicker::GetCoreTicker().RemoveTicker(TickHandle);
			TickHandle.Reset();
		}
		FPlatformMisc::RequestExit(false);
	}

	static bool TickTicker(float)
	{
		if (!GEngine || State.bFinished || State.Points.Num() < 2)
		{
			return true;
		}
		UWorld* PlayWorld = GEngine->GetCurrentPlayWorld();
		if (!PlayWorld)
		{
			return true;
		}

		const double Now = FPlatformTime::Seconds();
		if (Now - State.StartSeconds > MaxRunSeconds)
		{
			Finish(PlayWorld, TEXT("timeout"));
			return false;
		}

		APlayerController* PC = PlayWorld->GetFirstPlayerController();
		APawn* Van = PC ? PC->GetPawn() : nullptr;
		UChaosWheeledVehicleMovementComponent* Movement = Van ? Van->FindComponentByClass<UChaosWheeledVehicleMovementComponent>() : nullptr;
		if (!Van || !Movement)
		{
			return true;
		}

		const FVector Pos = Van->GetActorLocation();

		while (State.Waypoint < State.Points.Num() - 1 &&
			FVector::Dist2D(Pos, State.Points[State.Waypoint]) < WaypointRadiusCm)
		{
			++State.Waypoint;
		}

		const FVector Look = State.Points[FMath::Min(State.Waypoint + 3, State.Points.Num() - 1)];
		FVector ToTarget = Look - Pos;
		ToTarget.Z = 0.0;
		ToTarget.Normalize();
		FVector Forward = Van->GetActorForwardVector();
		Forward.Z = 0.0;
		Forward.Normalize();
		const float Cross = Forward.X * ToTarget.Y - Forward.Y * ToTarget.X;
		const float Dot = FMath::Max(0.2f, static_cast<float>(Forward.X * ToTarget.X + Forward.Y * ToTarget.Y));
		const float Steer = FMath::Clamp(2.2f * FMath::Sign(Cross) * FMath::Min(1.0f, FMath::Abs(Cross) / Dot * 1.4f), -1.0f, 1.0f);
		Movement->SetSteeringInput(Steer);

		const FVector Ahead = State.Points[FMath::Min(State.Waypoint + 14, State.Points.Num() - 1)];
		const float Curve = static_cast<float>(FVector::Dist2D(Ahead, Look));
		const float Speed = Movement->GetForwardSpeed();
		const float Want = Curve > 16000.0f ? 1400.0f : 900.0f;
		if (Speed < Want)
		{
			Movement->SetThrottleInput(Curve > 16000.0f ? 0.85f : 0.6f);
			Movement->SetBrakeInput(0.0f);
		}
		else
		{
			Movement->SetThrottleInput(0.0f);
			Movement->SetBrakeInput(0.4f);
		}
		Movement->SetHandbrakeInput(false);

		double OffCm = TNumericLimits<double>::Max();
		for (int32 i = FMath::Max(0, State.Waypoint - 6); i < FMath::Min(State.Points.Num() - 1, State.Waypoint + 12); ++i)
		{
			const FVector2D A(State.Points[i].X, State.Points[i].Y);
			const FVector2D B(State.Points[i + 1].X, State.Points[i + 1].Y);
			const FVector2D D = B - A;
			const float DD = D.SizeSquared();
			const FVector2D P(Pos.X, Pos.Y);
			const float T = DD > 0.0f ? FMath::Clamp(FVector2D::DotProduct(P - A, D) / DD, 0.0f, 1.0f) : 0.0f;
			OffCm = FMath::Min(OffCm, static_cast<double>(FVector2D::Distance(P, A + T * D)));
		}
		if (OffCm > OffroadLimitCm)
		{
			++State.OffroadSamples;
		}

		++State.Samples;
		State.SpeedSum += FMath::Abs(Speed);
		State.SpeedMax = FMath::Max(State.SpeedMax, FMath::Abs(Speed));

		if (FMath::Abs(Speed) < StallSpeedCm)
		{
			if (State.StallSince < 0.0)
			{
				State.StallSince = Now;
			}
			else if (Now - State.StallSince > StallTimeoutS)
			{
				Finish(PlayWorld, TEXT("stalled"));
				return false;
			}
		}
		else
		{
			State.StallSince = -1.0;
		}

		if (State.Waypoint >= State.Points.Num() - 2)
		{
			Finish(PlayWorld, TEXT("completed"));
			return false;
		}
		return true;
	}

	static bool LoadRoute()
	{
		const FString Path = FPaths::ProjectSavedDir() / TEXT("WorldGeneration/World.json");
		FString Json;
		if (!FFileHelper::LoadFileToString(Json, *Path))
		{
			UE_LOG(LogRoadToWisdom, Error, TEXT("[ROUTE] World.json not found at %s"), *Path);
			return false;
		}
		TSharedPtr<FJsonObject> Root;
		const TSharedRef<TJsonReader<>> Reader = TJsonReaderFactory<>::Create(Json);
		if (!FJsonSerializer::Deserialize(Reader, Root) || !Root.IsValid())
		{
			return false;
		}
		const TArray<TSharedPtr<FJsonValue>>* Roads = nullptr;
		if (!Root->TryGetArrayField(TEXT("roads"), Roads))
		{
			return false;
		}
		for (const TSharedPtr<FJsonValue>& RoadValue : *Roads)
		{
			const TSharedPtr<FJsonObject> Road = RoadValue->AsObject();
			FString Name;
			if (Road.IsValid() && Road->TryGetStringField(TEXT("name"), Name) && Name == TEXT("Main"))
			{
				const TArray<TSharedPtr<FJsonValue>>* Points = nullptr;
				if (Road->TryGetArrayField(TEXT("points"), Points))
				{
					for (const TSharedPtr<FJsonValue>& PointValue : *Points)
					{
						const TArray<TSharedPtr<FJsonValue>>* C = nullptr;
						if (PointValue->TryGetArray(C) && C->Num() >= 3)
						{
							State.Points.Emplace((*C)[0]->AsNumber() * 100.0, (*C)[1]->AsNumber() * 100.0, (*C)[2]->AsNumber() * 100.0);
						}
					}
				}
				break;
			}
		}
		State.RouteLengthCm = 0.0f;
		for (int32 i = 0; i + 1 < State.Points.Num(); ++i)
		{
			State.RouteLengthCm += FVector::Dist2D(State.Points[i], State.Points[i + 1]);
		}
		return State.Points.Num() > 2;
	}

	static void StartRouteDrive(const TArray<FString>& Args)
	{
		UGameEngine* Engine = GEngine ? Cast<UGameEngine>(GEngine) : nullptr;
		UWorld* World = Engine ? Engine->GetCurrentPlayWorld() : nullptr;
		if (!World)
		{
			UE_LOG(LogRoadToWisdom, Error, TEXT("[ROUTE] no play world"));
			return;
		}
		if (!LoadRoute())
		{
			UE_LOG(LogRoadToWisdom, Error, TEXT("[ROUTE] route load failed"));
			return;
		}

		APlayerController* PC = World->GetFirstPlayerController();
		if (!PC)
		{
			UE_LOG(LogRoadToWisdom, Error, TEXT("[ROUTE] no player controller"));
			return;
		}

		// Prefer the tagged placed van; otherwise the possessed default pawn drives.
		ARoadToWisdomPawn* Tagged = nullptr;
		for (TActorIterator<ARoadToWisdomPawn> It(World); It; ++It)
		{
			if (It->ActorHasTag(TEXT("QuietRoad.Van")))
			{
				Tagged = *It;
				break;
			}
		}
		if (Tagged)
		{
			PC->Possess(Tagged);
		}
		if (!PC->GetPawn())
		{
			UE_LOG(LogRoadToWisdom, Error, TEXT("[ROUTE] no pawn to drive"));
			return;
		}

		State.StartSeconds = FPlatformTime::Seconds();
		TickHandle = FTSTicker::GetCoreTicker().AddTicker(FTickerDelegate::CreateStatic(&TickTicker), 0.0f);

		UE_LOG(LogRoadToWisdom, Warning, TEXT("[ROUTE] driving %d points, %.0f m"), State.Points.Num(), RouteLengthM());
	}
}

static FAutoConsoleCommand GQRWRouteDrive(
	TEXT("QRW.RouteDrive"),
	TEXT("Drives the tagged van along the Main road route; writes Saved/WorldGeneration/VanRouteMain.json"),
	FConsoleCommandWithArgsDelegate::CreateStatic(&QRWRouteDrive::StartRouteDrive));

	float RouteY(float X)
	{
		const float T = FMath::Clamp(X / 24000.f, 0.f, 1.f);
		const float End = FMath::Clamp((X - 21000.f) / 3000.f, 0.f, 1.f);
		return 1200.f * FMath::Sin(2 * PI * T) + 350.f * FMath::Sin(4 * PI * T) - 900.f * End * End * (3 - 2 * End);
	}

	struct FQRDriveState
	{
		int32 Stage = 0;
		double Started = 0;
		float MaxDeviation = 0;
		bool bMovingExitTested = false;
		FVector SavedLocation = FVector::ZeroVector;
	};
}

bool FQRRouteSmoke::RunTest(const FString&)
{
	if (!FParse::Param(FCommandLine::Get(),TEXT("QuietRoadAutomation"))) {
		AddError(TEXT("Use -QuietRoadAutomation for the isolated slot."));
		return false;
	}
	UGameplayStatics::DeleteGameInSlot(TEXT("QuietRoad_Automation"), 0);
	auto State = MakeShared<FQRDriveState>();
	ADD_LATENT_AUTOMATION_COMMAND(FEditorLoadMap(TEXT("/Game/QuietRoad/Maps/L_QuietRoad_Prototype")));
	ADD_LATENT_AUTOMATION_COMMAND(FStartPIECommand(false));
	ADD_LATENT_AUTOMATION_COMMAND(FWaitLatentCommand(2.f));
	ADD_LATENT_AUTOMATION_COMMAND(
		FFunctionLatentCommand([this,State]()
			{
			UWorld* World=GEditor->PlayWorld;
			auto* PC=World ? Cast<ARoadToWisdomExplorerController>(UGameplayStatics::GetPlayerController(World,0)) : nullptr;
			if (!TestNotNull(TEXT("Route controller"),PC)) return true;
			auto* Van=PC->GetVanPawn();
			if (!TestNotNull(TEXT("Chaos camper"),Van)) return true;
			if (State->Stage==0)
			{
			auto* Character=Cast<ARoadToWisdomOnFootCharacter>(PC->GetPawn());
			if (!TestNotNull(TEXT("Starting pedestrian"),Character)) return true;
			FVector Door=Van->GetActorTransform().TransformPosition(FVector(120,-190,0));
			Door.Z=100;
			Character->SetActorLocation(Door);
			PC->RequestVehicleToggle();
			if (!TestTrue(TEXT("Enter before actual drive"),PC->IsDriving())) return true;
			State->Started=World->GetTimeSeconds();
			State->Stage=1;
			}
			const FVector Location=Van->GetActorLocation();
			const float Speed=Van->GetVelocity().Size();
			if (World->GetTimeSeconds()-State->Started > 100.f)
			{
			AddError(FString::Printf(TEXT("Route timed out at %s, speed %.1f"),*Location.ToString(),Speed));
			return true;
			}
			if (State->Stage==1)
			{
			State->MaxDeviation=FMath::Max(State->MaxDeviation,FMath::Abs(Location.Y-RouteY(Location.X)));
			const float TargetX=Location.X+550.f;
			const FVector Target(TargetX,RouteY(TargetX),Location.Z);
			const FVector Local=Van->GetActorTransform().InverseTransformPosition(Target);
			Van->DoSteering(FMath::Clamp(FMath::Atan2(Local.Y,Local.X)*1.7f,-0.7f,0.7f));
			if (Speed > 700.f) Van->DoBrake(.1f);
			else Van->DoThrottle(.24f);
			if (!State->bMovingExitTested && Speed > 300.f)
			{
			PC->RequestVehicleToggle();
			TestTrue(TEXT("Exit rejected while moving"),PC->IsDriving());
			State->bMovingExitTested=true;
			}
			if (Location.X < 23200.f) return false;
			State->Stage=2;
			}
			Van->DoThrottle(0.f);
			Van->DoBrake(1.f);
			if (Speed > 25.f) return false;
			Van->DoHandbrakeStart();
			PC->RequestVehicleToggle();
			TestFalse(TEXT("Exit after braking at relay"),PC->IsDriving());
			TestTrue(TEXT("Entire route within road clearance"),State->MaxDeviation < 450.f);
			for (TActorIterator<ARoadToWisdomMissionManager> I(World);I;++I)
			{
			TestEqual(TEXT("Physical arrival advances mission"),I->GetStage(),EQRRelayMissionStage::FindFuse);
			PC->RequestVehicleToggle();
			TestTrue(TEXT("Re-enter at relay"),PC->IsDriving());
			auto* Save=World->GetGameInstance()->GetSubsystem<URoadToWisdomSaveCoordinator>();
			TestTrue(TEXT("Save while parked in driving mode"),Save->SaveSession(*I));
			}
			State->SavedLocation=Van->GetActorLocation();
			for (TActorIterator<ARoadToWisdomVanRadioPanel> I(World);I;++I)
			TestTrue(TEXT("Radio travelled with camper"),FVector::Dist(I->GetActorLocation(),State->SavedLocation)<400.f);
			AddInfo(FString::Printf(TEXT("Chaos route driven to %s; maximum centre deviation %.1f cm."),
				*State->SavedLocation.ToString(),State->MaxDeviation));
			return true;
			})
	);
	ADD_LATENT_AUTOMATION_COMMAND(FEndPlayMapCommand());
	ADD_LATENT_AUTOMATION_COMMAND(FStartPIECommand(false));
	ADD_LATENT_AUTOMATION_COMMAND(FWaitLatentCommand(2.f));
	ADD_LATENT_AUTOMATION_COMMAND(
		FFunctionLatentCommand([this,State]()
			{
			auto* PC=Cast<ARoadToWisdomExplorerController>(UGameplayStatics::GetPlayerController(GEditor->PlayWorld,0));
			if (!TestNotNull(TEXT("Reloaded controller"),PC)) return true;
			TestTrue(TEXT("Driving possession restored"),PC->IsDriving());
			if (auto* Van=PC->GetVanPawn())
			{
			TestTrue(TEXT("Vehicle restored near relay"),FVector::Dist(Van->GetActorLocation(),State->SavedLocation)<100.f);
			TestEqual(TEXT("Restored controller possesses persistent camper"),static_cast<APawn*>(PC->GetPawn()),static_cast<APawn*>(Van));
			}
			UGameplayStatics::DeleteGameInSlot(TEXT("QuietRoad_Automation"),0);
			return true;
			})
	);
	ADD_LATENT_AUTOMATION_COMMAND(FEndPlayMapCommand());
	return true;
}

void ARoadToWisdomPawn::DoBrakeStop()
{
	// call the Blueprint hook for the brake lights
	BrakeLights(false);

	// reset brake input to zero
	ChaosVehicleMovement->SetBrakeInput(0.0f);
}

void ARoadToWisdomPawn::DoHandbrakeStart()
{
	// add the input
	ChaosVehicleMovement->SetHandbrakeInput(true);

	// call the Blueprint hook for the break lights
	BrakeLights(true);
}

void ARoadToWisdomPawn::DoHandbrakeStop()
{
	// add the input
	ChaosVehicleMovement->SetHandbrakeInput(false);

	// call the Blueprint hook for the break lights
	BrakeLights(false);
}

void ARoadToWisdomPawn::DoLookAround(float YawDelta)
{
	// rotate the spring arm
	BackSpringArm->AddLocalRotation(FRotator(0.0f, YawDelta, 0.0f));
}

void ARoadToWisdomPawn::DoToggleCamera()
{
	// toggle the active camera flag
	bFrontCameraActive = !bFrontCameraActive;

	FrontCamera->SetActive(bFrontCameraActive);
	BackCamera->SetActive(!bFrontCameraActive);
}

void ARoadToWisdomPawn::DoResetVehicle()
{
	// reset to a location slightly above our current one
	FVector ResetLocation = GetActorLocation() + FVector(0.0f, 0.0f, 50.0f);

	// reset to our yaw. Ignore pitch and roll
	FRotator ResetRotation = GetActorRotation();
	ResetRotation.Pitch = 0.0f;
	ResetRotation.Roll = 0.0f;

	// teleport the actor to the reset spot and reset physics
	SetActorTransform(FTransform(ResetRotation, ResetLocation, FVector::OneVector), false, nullptr, ETeleportType::TeleportPhysics);

	GetMesh()->SetPhysicsAngularVelocityInDegrees(FVector::ZeroVector);
	GetMesh()->SetPhysicsLinearVelocity(FVector::ZeroVector);
}

void ARoadToWisdomPawn::FlippedCheck()
{
	// check the difference in angle between the mesh's up vector and world up
	const float UpDot = FVector::DotProduct(FVector::UpVector, GetMesh()->GetUpVector());

	if (UpDot < FlipCheckMinDot)
	{
		// is this the second time we've checked that the vehicle is still flipped?
		if (bPreviousFlipCheck)
		{
			// reset the vehicle to upright
			DoResetVehicle();
		}
		
		// set the flipped check flag so the next check resets the car
		bPreviousFlipCheck = true;

	} else {

		// we're upright. reset the flipped check flag
		bPreviousFlipCheck = false;
	}
}

void ARoadToWisdomPlayerController::BeginPlay()
{
	Super::BeginPlay();
	
	// ensure we're attached to the vehicle pawn so that World Partition streaming works correctly
	bAttachToPawn = true;
}

void ARoadToWisdomPlayerController::SetupInputComponent()
{
	Super::SetupInputComponent();
	
	// only add IMCs for local player controllers
	if (IsLocalPlayerController())
	{
		// Add Input Mapping Contexts
		if (UEnhancedInputLocalPlayerSubsystem* Subsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(GetLocalPlayer()))
		{
			for (UInputMappingContext* CurrentContext : DefaultMappingContexts)
			{
				Subsystem->AddMappingContext(CurrentContext, 0);
			}

			// only add these IMCs if we're not using mobile touch input
			if (!ShouldUseTouchControls())
			{
				for (UInputMappingContext* CurrentContext : MobileExcludedMappingContexts)
				{
					Subsystem->AddMappingContext(CurrentContext, 0);
				}
			}

			if (bUseSteeringWheelControls)
			{
				Subsystem->AddMappingContext(SteeringWheelInputMappingContext, 0);
			}
		}
	}

	// only spawn UI on local player controllers
	if (IsLocalPlayerController())
	{
		if (ShouldUseTouchControls())
		{
			// spawn the mobile controls widget
			MobileControlsWidget = CreateWidget<UUserWidget>(this, MobileControlsWidgetClass);

			if (MobileControlsWidget)
			{
				// add the controls to the player screen
				MobileControlsWidget->AddToPlayerScreen(0);

			} else {

				UE_LOG(LogRoadToWisdom, Error, TEXT("Could not spawn mobile controls widget."));

			}
		}
		

		// spawn the UI widget and add it to the viewport
		VehicleUI = CreateWidget<URoadToWisdomUI>(this, VehicleUIClass);

		if (VehicleUI)
		{
			VehicleUI->AddToPlayerScreen(0);

		} else {

			UE_LOG(LogRoadToWisdom, Error, TEXT("Could not spawn vehicle UI widget."));

		}
	}
}

void ARoadToWisdomPlayerController::Tick(float Delta)
{
	Super::Tick(Delta);

	if (IsValid(VehiclePawn) && IsValid(VehicleUI))
	{
		VehicleUI->UpdateSpeed(VehiclePawn->GetChaosVehicleMovement()->GetForwardSpeed());
		VehicleUI->UpdateGear(VehiclePawn->GetChaosVehicleMovement()->GetCurrentGear());
	}
}

void ARoadToWisdomPlayerController::OnPossess(APawn* InPawn)
{
	Super::OnPossess(InPawn);

	// get a pointer to the controlled pawn
	VehiclePawn = CastChecked<ARoadToWisdomPawn>(InPawn);

	// subscribe to the pawn's OnDestroyed delegate
	VehiclePawn->OnDestroyed.AddDynamic(this, &ARoadToWisdomPlayerController::OnPawnDestroyed);
}

void ARoadToWisdomPlayerController::OnPawnDestroyed(AActor* DestroyedPawn)
{
	// find the player start
	TArray<AActor*> ActorList;
	UGameplayStatics::GetAllActorsOfClass(GetWorld(), APlayerStart::StaticClass(), ActorList);

	if (ActorList.Num() > 0)
	{
		// spawn a vehicle at the player start
		const FTransform SpawnTransform = ActorList[0]->GetActorTransform();

		if (ARoadToWisdomPawn* RespawnedVehicle = GetWorld()->SpawnActor<ARoadToWisdomPawn>(VehiclePawnClass, SpawnTransform))
		{
			// possess the vehicle
			Possess(RespawnedVehicle);
		}
	}
}

bool ARoadToWisdomPlayerController::ShouldUseTouchControls() const
{
	// are we on a mobile platform? Should we force touch?
	return SVirtualJoystick::ShouldDisplayTouchInterface() || bForceTouchControls;
}

void ATimeTrialPlayerController::SetupInputComponent()
{
	Super::SetupInputComponent();

	// only add IMCs for local player controllers
	if (IsLocalPlayerController())
	{
		// Add Input Mapping Contexts
		if (UEnhancedInputLocalPlayerSubsystem* Subsystem = ULocalPlayer::GetSubsystem<UEnhancedInputLocalPlayerSubsystem>(GetLocalPlayer()))
		{
			for (UInputMappingContext* CurrentContext : DefaultMappingContexts)
			{
				Subsystem->AddMappingContext(CurrentContext, 0);
			}

			// only add these IMCs if we're not using mobile touch input
			if (!ShouldUseTouchControls())
			{
				for (UInputMappingContext* CurrentContext : MobileExcludedMappingContexts)
				{
					Subsystem->AddMappingContext(CurrentContext, 0);
				}
			}

			if (bUseSteeringWheelControls)
			{
				Subsystem->AddMappingContext(SteeringWheelInputMappingContext, 0);
			}
		}
	}

	// only spawn UI on local player controllers
	if (IsLocalPlayerController())
	{
		if (ShouldUseTouchControls())
		{
			// spawn the mobile controls widget
			MobileControlsWidget = CreateWidget<UUserWidget>(this, MobileControlsWidgetClass);

			if (MobileControlsWidget)
			{
				// add the controls to the player screen
				MobileControlsWidget->AddToPlayerScreen(0);

			} else {

				UE_LOG(LogRoadToWisdom, Error, TEXT("Could not spawn mobile controls widget."));

			}
		}

		// create the UI widget
		UIWidget = CreateWidget<UTimeTrialUI>(this, UIWidgetClass);

		if (UIWidget)
		{
			UIWidget->AddToPlayerScreen(0);

			// subscribe to the race start delegate
			UIWidget->OnRaceStart.AddDynamic(this, &ATimeTrialPlayerController::StartRace);

		} else {

			UE_LOG(LogRoadToWisdom, Error, TEXT("Could not spawn Time Trial UI widget."));

		}
		

		// spawn the UI widget and add it to the viewport
		VehicleUI = CreateWidget<URoadToWisdomUI>(this, VehicleUIClass);

		if (VehicleUI)
		{
			VehicleUI->AddToPlayerScreen(0);

		} else {

			UE_LOG(LogRoadToWisdom, Error, TEXT("Could not spawn vehicle UI widget."));

		}
	}
}

void ATimeTrialPlayerController::OnPossess(APawn* InPawn)
{
	Super::OnPossess(InPawn);

	// get a pointer to the controlled pawn
	VehiclePawn = CastChecked<ARoadToWisdomPawn>(InPawn);

	// subscribe to the pawn's OnDestroyed delegate
	VehiclePawn->OnDestroyed.AddDynamic(this, &ATimeTrialPlayerController::OnPawnDestroyed);

	// disable input on the pawn if the race hasn't started yet
	if (!bRaceStarted)
	{
		VehiclePawn->DisableInput(this);
	}	
}

{
	Super::Tick(Delta);

	if (IsValid(VehiclePawn) && IsValid(VehicleUI))
	{
		VehicleUI->UpdateSpeed(VehiclePawn->GetChaosVehicleMovement()->GetForwardSpeed());
		VehicleUI->UpdateGear(VehiclePawn->GetChaosVehicleMovement()->GetCurrentGear());
	}
}

void ATimeTrialPlayerController::StartRace()
{
	// get the finish line from the game mode
	if (ATimeTrialGameMode* GM = Cast<ATimeTrialGameMode>(GetWorld()->GetAuthGameMode()))
	{
		SetTargetGate(GM->GetFinishLine()->GetNextMarker());
	}

	// raise the race started flag so any respawned vehicles start with controls unlocked 
	bRaceStarted = true;

	// start the first lap
	CurrentLap = 0;
	IncrementLapCount();

	// enable input on the pawn
	if (GetPawn())
	{
		GetPawn()->EnableInput(this);
	}
}

void ATimeTrialPlayerController::IncrementLapCount()
{
	// increment the lap counter
	++CurrentLap;

	// update the UI
	UIWidget->UpdateLapCount(CurrentLap, GetWorld()->GetTimeSeconds());
}

void ATimeTrialPlayerController::OnPawnDestroyed(AActor* DestroyedPawn)
{
	// find the player start
	TArray<AActor*> ActorList;
	UGameplayStatics::GetAllActorsOfClass(GetWorld(), APlayerStart::StaticClass(), ActorList);

	if (ActorList.Num() > 0)
	{
		// spawn a vehicle at the player start
		const FTransform SpawnTransform = ActorList[0]->GetActorTransform();

		if (ARoadToWisdomPawn* RespawnedVehicle = GetWorld()->SpawnActor<ARoadToWisdomPawn>(VehiclePawnClass, SpawnTransform))
		{
			// possess the vehicle
			Possess(RespawnedVehicle);
		}
	}
}

{
	// are we on a mobile platform? Should we force touch?
	return SVirtualJoystick::ShouldDisplayTouchInterface() || bForceTouchControls;
}

void UTimeTrialUI::UpdateLapCount(int32 Lap, float NewLapStartTime)
{
	// save the new lap start time
	LapStartTime = NewLapStartTime;

	// calculate the lap time
	const float LapTime = NewLapStartTime - LastLapTime;

	// is this the first lap?
	if (Lap > 1)
	{
		// do we have an invalid lap time?
		if (BestLapTime < 0.0f)
		{
			// save the current lap time
			BestLapTime = LapTime;

		} else {

			// not the first lap: do we have a lower lap time?
			if (LapTime < BestLapTime)
			{
				// save the best lap time
				BestLapTime = LapTime;
			}

		}
		
	} else {

		// first lap: save an invalid lap time
		BestLapTime = -1.0f;

	}

	// save the current lap
	CurrentLap = Lap;

	// save the lap start time
	LastLapTime = NewLapStartTime;

	// pass control to BP to update the widgets
	BP_UpdateLaps();
}

void UTimeTrialUI::StartRace()
{
	// broadcast the delegate
	OnRaceStart.Broadcast();
}
